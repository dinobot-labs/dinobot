import { chromium } from "playwright-core";
const EDGE = "C:/Program Files (x86)/Microsoft/Edge/Application/msedge.exe";
const page_html = "<html><body style='margin:0'><div id=b style='width:200px;height:200px;background:#19CA9D;animation:m 1s linear infinite alternate'></div><style>@keyframes m{to{transform:translateX(600px) rotate(90deg)}}</style></body></html>";
async function measure(args, headless) {
  const browser = await chromium.launch({ executablePath: EDGE, headless, args });
  const page = await browser.newPage({ viewport: { width: 1440, height: 900 } });
  await page.setContent(page_html);
  const res = await page.evaluate(() => new Promise((resolve) => {
    const ts = []; const start = performance.now();
    function f(t) { ts.push(t); if (t - start < 2000) requestAnimationFrame(f); else resolve(ts); }
    requestAnimationFrame(f);
  }));
  const gpu = await page.evaluate(() => { const c = document.createElement("canvas"); const gl = c.getContext("webgl"); if (!gl) return "no-webgl"; const d = gl.getExtension("WEBGL_debug_renderer_info"); return d ? gl.getParameter(d.UNMASKED_RENDERER_WEBGL) : "unknown"; });
  await browser.close();
  const iv = []; for (let i = 1; i < res.length; i++) iv.push(res[i] - res[i-1]);
  iv.sort((a,b)=>a-b);
  return { frames: res.length, fps: (res.length - 1) / ((res[res.length-1] - res[0]) / 1000), p50: iv[Math.floor(iv.length*0.5)], p99: iv[Math.floor(iv.length*0.99)], gpu };
}
console.log("default headless", JSON.stringify(await measure([], true)));
console.log("uncapped headless", JSON.stringify(await measure(["--disable-gpu-vsync", "--disable-frame-rate-limit"], true)));
