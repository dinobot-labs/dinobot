#!/usr/bin/env bun
// dinobot slack-setup: restart a TUI-owned dinobot daemon once no thread is busy, inject credentials from files,
// and confirm that the new daemon's boot line lists the expected channel.
//
// Run it OUTSIDE the daemon's process tree (Windows: scripts/detach-windows.ps1; Linux/macOS: setsid or
// systemd-run --user). Started from a bash tool call it would die together with the daemon it stops.
// Your own conversation counts as busy, so the restart happens after you end your turn.
//
//   bun restart-when-idle.mjs [--api http://127.0.0.1:8786]
//       [--env NAME=<file>]...               e.g. --env DINOBOT_SLACK_BOT_TOKEN=/path/bot-token (repeatable)
//       [--expect-channel slack]             boot line must list this channel (default slack)
//       [--rollback <manifest backup>]       restore this manifest and start again if the expected boot fails
//       [--notify-token-file <xoxb file> --notify-channel <D...|C...>]   post the outcome to Slack
//       [--manifest <path>] [--log <path>] [--dinobot <command>]       overrides; normally discovered
//       [--max-wait-hours 6] [--dry-run]
//
// Writes restart-result.json and appends restart-trace.log next to the manifest. Never prints secrets.
import { appendFileSync, closeSync, copyFileSync, existsSync, openSync, readFileSync, statSync, writeFileSync } from "node:fs";
import { dirname, join } from "node:path";

const opts = { api: "http://127.0.0.1:8786", env: {}, expect: "slack", maxWaitHours: 6, dryRun: false };
const argv = process.argv.slice(2);
for (let i = 0; i < argv.length; i++) {
	const flag = argv[i];
	const value = () => {
		const v = argv[++i];
		if (v === undefined) throw new Error("missing value for " + flag);
		return v;
	};
	if (flag === "--api") opts.api = value().replace(/\/$/, "");
	else if (flag === "--env") {
		const pair = value(), at = pair.indexOf("=");
		if (at < 1) throw new Error("--env takes NAME=<file>");
		opts.env[pair.slice(0, at)] = pair.slice(at + 1);
	} else if (flag === "--expect-channel") opts.expect = value();
	else if (flag === "--rollback") opts.rollback = value();
	else if (flag === "--notify-token-file") opts.notifyToken = value();
	else if (flag === "--notify-channel") opts.notifyChannel = value();
	else if (flag === "--manifest") opts.manifest = value();
	else if (flag === "--log") opts.log = value();
	else if (flag === "--dinobot") opts.dinobot = value();
	else if (flag === "--max-wait-hours") opts.maxWaitHours = Number(value());
	else if (flag === "--dry-run") opts.dryRun = true;
	else throw new Error("unknown argument " + flag);
}
if (!/^http:\/\/(127\.0\.0\.1|localhost)(:\d+)?$/.test(opts.api)) throw new Error("--api must be the loopback client API");

const sleep = (ms) => new Promise((resolve) => setTimeout(resolve, ms));
async function daemon() {
	try {
		const response = await fetch(opts.api + "/v1/daemon", { signal: AbortSignal.timeout(5000) });
		return response.ok ? await response.json() : { error: "GET /v1/daemon: HTTP " + response.status };
	} catch {
		return null;
	}
}

const first = await daemon();
const manifest = opts.manifest ?? first?.manifestPath;
if (!manifest) throw new Error("no daemon answers at " + opts.api + "; pass --manifest");
const here = dirname(manifest);
const resultPath = join(here, "restart-result.json"), tracePath = join(here, "restart-trace.log");
function trace(line) {
	const text = new Date().toISOString() + " " + line;
	if (opts.dryRun) console.log(text);
	else appendFileSync(tracePath, text + "\n");
}

let log = opts.log;
if (!log) {
	const parsed = typeof Bun !== "undefined" && Bun.YAML ? Bun.YAML.parse(readFileSync(manifest, "utf8")) : null;
	const dataDir = parsed?.bot?.dataDir;
	if (typeof dataDir !== "string") throw new Error("cannot read bot.dataDir from " + manifest + "; pass --log");
	log = join(dataDir, "dinobot.log");
}
const command = opts.dinobot ? [opts.dinobot] : (() => {
	const found = typeof Bun !== "undefined" ? Bun.which("dinobot") : null;
	if (!found) throw new Error("dinobot is not on PATH; pass --dinobot <path>");
	return [found];
})();
const secrets = {};
for (const [name, file] of Object.entries(opts.env)) {
	const value = readFileSync(file, "utf8").trim();
	if (!value || /\s/.test(value)) throw new Error(name + ": " + file + " must hold one value");
	secrets[name] = value;
}

async function notify(text) {
	if (!opts.notifyToken || !opts.notifyChannel) return;
	try {
		const token = readFileSync(opts.notifyToken, "utf8").trim();
		const response = await fetch("https://slack.com/api/chat.postMessage", {
			method: "POST",
			headers: { authorization: "Bearer " + token, "content-type": "application/json; charset=utf-8" },
			body: JSON.stringify({ channel: opts.notifyChannel, text }),
		});
		const body = await response.json();
		trace("notify ok=" + body.ok + (body.error ? " error=" + body.error : ""));
	} catch (error) {
		trace("notify failed " + error);
	}
}
function finish(result, code) {
	writeFileSync(resultPath, JSON.stringify({ at: new Date().toISOString(), ...result }, null, 1));
	trace("result " + JSON.stringify(result));
	process.exit(code);
}
function spawnDaemon() {
	const fd = openSync(log, "a");
	try {
		const child = Bun.spawn([...command, "serve", "--config", manifest], {
			stdio: ["ignore", fd, fd],
			detached: true,
			// TUI-owned like a TUI-spawned daemon, so a TUI or this script can restart it again later.
			env: { ...process.env, ...secrets, DINOBOT_DAEMON_OWNER: "tui" },
		});
		child.unref();
		return child.pid;
	} finally {
		closeSync(fd);
	}
}
async function bootLine(offset, ms) {
	const end = Date.now() + ms;
	let lastError = null;
	while (Date.now() < end) {
		await sleep(1000);
		if (!existsSync(log)) continue;
		const lines = readFileSync(log).subarray(offset).toString("utf8").split("\n");
		const up = lines.find((line) => line.startsWith("dinobot up:"));
		if (up) return { up };
		lastError = lines.filter((line) => line.startsWith("error: ")).pop() ?? lastError;
	}
	return { up: null, error: lastError ?? "no boot line within " + ms / 1000 + "s" };
}
const listsChannel = (up) => (/ channels=([^ ]+)/.exec(up)?.[1] ?? "").split(",").includes(opts.expect);

trace("helper pid=" + process.pid + " manifest=" + manifest + " log=" + log + " command=" + command.join(" ") + " env=" + Object.keys(secrets).join(","));
if (opts.dryRun) {
	trace("daemon " + JSON.stringify(first && { pid: first.pid, version: first.version, owner: first.owner, busy: first.busy }));
	trace("rollback " + (opts.rollback ? (existsSync(opts.rollback) ? "exists" : "MISSING") : "none"));
	process.exit(0);
}

const started = Date.now();
for (let info = first, quiet = 0; ; info = await daemon()) {
	if (!info) {
		trace("no daemon answering; starting directly");
		break;
	}
	if (info.error) trace(info.error);
	else if (Array.isArray(info.busy) && info.busy.length === 0) {
		const response = await fetch(opts.api + "/v1/daemon/shutdown", {
			method: "POST",
			headers: { "content-type": "application/json" },
			body: JSON.stringify({ pid: info.pid }),
		}).catch(() => null);
		if (response?.status === 202) {
			trace("shutdown accepted pid=" + info.pid + " version=" + info.version);
			break;
		}
		const body = response ? await response.text() : "no response";
		trace("shutdown refused " + (response?.status ?? "-") + " " + body);
		if (body.includes("external")) {
			await notify("⚠️ dinobot restart skipped: this daemon is managed outside the TUI (restart it with its service manager).");
			finish({ ok: false, reason: "daemon is not TUI-owned; restart it through its supervisor" }, 1);
		}
	} else if (Date.now() - started > opts.maxWaitHours * 3600000) {
		await notify("⚠️ dinobot restart skipped: threads stayed busy for " + opts.maxWaitHours + "h.");
		finish({ ok: false, reason: "threads busy for " + opts.maxWaitHours + "h", busy: info.busy }, 1);
	} else if (quiet++ % 40 === 0) trace("waiting for idle; busy=" + JSON.stringify(info.busy));
	await sleep(15000);
}

for (let i = 0; i < 60 && (await daemon()); i++) await sleep(1000);
if (await daemon()) {
	await notify("⚠️ dinobot restart failed: the old daemon did not stop within 60s.");
	finish({ ok: false, reason: "old daemon still answering 60s after shutdown" }, 1);
}
await sleep(2000);
let offset = existsSync(log) ? statSync(log).size : 0;
trace("spawned pid=" + spawnDaemon());
const boot = await bootLine(offset, 90000);
const now = await daemon();
if (boot.up && listsChannel(boot.up) && now) {
	await notify("✅ dinobot restarted with the " + opts.expect + " channel (" + now.version + "). Reopen the TUI if it disconnected.");
	finish({ ok: true, boot: boot.up, pid: now.pid, version: now.version }, 0);
}
trace("expected boot not confirmed: " + JSON.stringify(boot) + " answering=" + Boolean(now));
if (now || !opts.rollback) {
	await notify("⚠️ dinobot restarted but the " + opts.expect + " channel was not confirmed: " + (boot.error ?? boot.up));
	finish({ ok: false, reason: "expected channel not confirmed", boot, answering: Boolean(now) }, 1);
}
copyFileSync(manifest, manifest + ".failed");
copyFileSync(opts.rollback, manifest);
offset = existsSync(log) ? statSync(log).size : 0;
trace("rollback spawned pid=" + spawnDaemon());
const fallback = await bootLine(offset, 90000);
await notify("⚠️ dinobot could not start with the new " + opts.expect + " settings, so it was restarted with the previous manifest. Error: " + (boot.error ?? "unknown"));
finish({ ok: false, reason: "new manifest failed; previous manifest restored (failed copy: " + manifest + ".failed)", error: boot.error, fallbackBoot: fallback.up ?? fallback.error }, 1);
